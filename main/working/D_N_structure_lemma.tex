\begin{equation}
	\begin{gathered}
		\wt(x)=\frac1{N-1}, \wt(y)=\frac{N-2}{2(N-1)}\\
		\wt(e_r)=\frac r{N-1}, \wt(h)=\frac{N-2}{2(N-1)}\\
		\wt(s_r)=1-\frac r{N-1}, \wt(\tau)=\frac{N}{2(N-1)}\\
		\wt(\alpha_k)=1-\frac{k}{N-1}=\wt(s_k) \qquad
		\wt(\beta) = \frac{N}{2(N-1)} = \wt(\tau)
	\end{gathered}
	\label{eq:weightDN}
\end{equation}
\begin{lem}
	Пусть дано некоторое многообразие Дубровина-Фробениуса \((M, \circ)\) размерности \(\mu \geq 5\) с градуировкой \(D_\mu\). Тогда.
	\begin{enumerate}
		\item Существует семейство фактор алгебр, таких что 
		\[T_pM \simeq B = \mathbb{C}[x, y]/\left(x^{\mu - 1}  - \sum_{k=0}^{\mu - 2}\alpha_k (s, \tau)  x^k - \beta(s, \tau) y \right)\]
		\item Для любого изоморфизма градуированных алгебр \(\psi : T_0M \to B|_0\) выполнено \[\psi\left(\tang[s^a]\right) = c_i x^a, \psi\left(\tang[\tau]\right) = c_\tau y \qquad c_i, c_\tau \in \mathbb{C}\]
	\end{enumerate}
\end{lem}
